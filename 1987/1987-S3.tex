    \item Rearranging the first equation for \(y\) and substituting into the 2nd equation gives us:

\begin{equation*}
\frac{1}{5} \left( \frac{dx}{dt} + 2x \right) = y
\end{equation*}

\begin{equation*}
\frac{dy}{dt} + x - 2y = 2 \cos t
\end{equation*}

\begin{equation*}
\frac{1}{5} (\ddot{x} + 2\dot{x}) + x - 2\left(\frac{1}{5}(\dot{x} + 2x)\right) = 2 \cos t
\end{equation*}
\begin{equation*}
\frac{1}{5} \ddot{x} + \frac{2}{5} \dot{x} + x - \frac{2}{5} \dot{x} - \frac{4}{5} x = 2 \cos t
\end{equation*}
\begin{equation*}
\ddot{x} + 2\dot{x} + 5x - 2\dot{x} - 4x = 10 \cos t
\end{equation*}
\begin{equation*}
\ddot{x} + x = 10 \cos t
\end{equation*}

\noindent The complementary function satisfies
\begin{equation*}
\ddot{x}_{CF} = -x_{CF} \implies x_{CF} = A \cos(t) + B \sin(t)
\end{equation*}

\noindent For the particular integral, we anticipate a solution of the form
\begin{align*}
x &= Ct \sin t \\
\dot{x} &= C \sin t + Ct \cos t \\
\ddot{x} &= 2C \cos t -  Ct \sin t
\end{align*}
We thus choose \(C=5\). The general solution is then
\[
x(t)= A\cos{t}+B\sin{t} +5t\sin{t}
\]
Applying the initial condition \(x(0)=0\) gives \(A=0\). Now substituting this into the equation for \(y\) gives us
\begin{align*}
y(t) = \frac{1}{5}\Big( B\cos t + (5+2B)\sin t + 5t\cos t +10t\sin t \Big)
\end{align*}
Differentiating yields
\[
\dot{y}(t) = \frac{1}{5}\Big( (10-B)\sin t + (10+2B)\cos t  -5t\sin t +10t\cos t \Big)
\]
Applying the initial condition \(\dot{y}(0)=0\) we obtain
\(
10+2B=0.
\)
Thus \(B=-5\). The solution is thus
\begin{align*}
    x(t) &= 5t\sin t - 5 \sin t \\
    y(t) &= 3\sin t - t\sin t + 2t\cos t
\end{align*}
Note that if instead we have \(a>1\) we will obtain the equation
\[
\ddot{x} + (5a-4)x = 10\cos t
\]
Thus the general solution for the complementary function will instead take the form \(x(t)=A\cos{\omega t}+ B\sin{\omega t}\) for some \(\omega>1\). Similarly, the particular integral will instead take the form \(x(t)=C\cos{t}\).
